BoolExpr ::=
      "true"
    | "false"
    | "and" "(" BoolExpr "," BoolExpr ")"
    | "or" "(" BoolExpr "," BoolExpr ")"
    | "not" "(" BoolExpr ")"
    | "implies" "(" BoolExpr "," BoolExpr ")"
    | "iff" "(" BoolExpr "," BoolExpr ")"
    | "=" "(" ArithExpr "," ArithExpr ")"
    | "!=" "(" ArithExpr "," ArithExpr ")"
    | "<" "(" ArithExpr "," ArithExpr ")"
    | "<=" "(" ArithExpr "," ArithExpr ")"
    | ">" "(" ArithExpr "," ArithExpr ")"
    | ">=" "(" ArithExpr "," ArithExpr ")"

ArithExpr ::=
      Integer
    | Variable
    | "+" "(" ArithExpr "," ArithExpr ")"
    | "-" "(" ArithExpr "," ArithExpr ")"
    | "neg" "(" ArithExpr ")"
    | "*" "(" ArithExpr "," ArithExpr ")"
    | "div" "(" ArithExpr "," ArithExpr ")"
    | "mod" "(" ArithExpr "," ArithExpr ")"
    | "pow" "(" ArithExpr "," NonNegativeInteger ")"
    | "abs" "(" ArithExpr ")"
    | "min" "(" ArithExpr "," ArithExpr ")"
    | "max" "(" ArithExpr "," ArithExpr ")"
    | "ite" "(" BoolExpr "," ArithExpr "," ArithExpr ")"

RankingExpr ::=
      ArithExpr
    | "lex" "(" ArithExpr "," RankingExpr ")"

Variable ::= any variable name provided for the target loop
Integer ::= any mathematical integer literal
NonNegativeInteger ::= any integer literal k such that k >= 0
